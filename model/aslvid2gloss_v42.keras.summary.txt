from keras.src.activations.activations import ReLU
from keras.src.layers import Flatten, Reshape, Dense, Input, TimeDistributed, MultiHeadAttention
from keras.src.activations import softmax
from numpy import float32, float64


from .lmark_constant import LANDMARK_SHAPE, QUANTITY_FRAME, LEN_GLOSS, QUANTITY_HAND_LMARK


data_in= Input(
    shape=(QUANTITY_FRAME, LANDMARK_SHAPE[0], LANDMARK_SHAPE[1]),
    dtype=float32,
    name='batch_vid',
)
x= Reshape(
    target_shape=(QUANTITY_FRAME, -1),
)(data_in) # now shape be (22, 86*2) or (18, 172)




x= TimeDistributed(Dense(
    units=LANDMARK_SHAPE[0],
    activation=ReLU(negative_slope=0.0, max_value=256.0, threshold=0.0),
    dtype=float64
))(x)
x= MultiHeadAttention(
    num_heads=8,
    key_dim=LANDMARK_SHAPE[0],
    dropout=0.05,
    use_bias=True,
)(
    query=x,
    value=x,
    key=x,
)
x= TimeDistributed(Dense(
    units=QUANTITY_HAND_LMARK*2,
    activation=ReLU(negative_slope=0.0, max_value=256.0, threshold=0.0),
    dtype=float64
))(x)
x= MultiHeadAttention(
    num_heads=8,
    key_dim=QUANTITY_HAND_LMARK*2,
    value_dim=QUANTITY_HAND_LMARK*2,
    dropout=0.05,
    use_bias=True,
)(
    query=x,
    value=x,
    key=x,
)




ann= Flatten()(x)
ann= Dense(
    units=LANDMARK_SHAPE[0]*2,
    activation=ReLU(negative_slope=0.0, max_value=256.0, threshold=0.0),
    dtype=float64,
)(ann)

from keras.src.losses import sparse_categorical_crossentropy

        inputs=data_in,
        outputs=data_out
    )
    model.compile(
        optimizer=Adam(learning_rate=0.0001),
        loss=sparse_categorical_crossentropy,
        metrics=['accuracy']

        x=get_data_landmark(train_val=KEY_TRAIN),
